{{- /* Extract direct child output while discarding author-formatting whitespace. */ -}} {{- $source := printf "%s" .source -}} {{- $start := "" -}} {{- $end := "" -}} {{- $pattern := printf `(?s)%s.*?%s` $start $end -}} {{- $matches := findRE $pattern $source -}} {{- $remaining := strings.TrimSpace (replaceRE $pattern "" $source) -}} {{- if $remaining -}} {{- errorf "shortcode %q accepts filetree/folder and filetree/file children only at %s" .name .position -}} {{- end -}} {{- $children := slice -}} {{- range $matches -}} {{- $child := strings.TrimPrefix $start . | strings.TrimSuffix $end | strings.TrimSpace -}} {{- $children = $children | append $child -}} {{- end -}} {{- return (delimit $children "\n") -}}
