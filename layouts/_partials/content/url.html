{{- $name := .name | default "content component" -}} {{- $parameter := .parameter | default "URL" -}} {{- $position := .position | default "unknown position" -}} {{- $kind := .kind | default "link" -}} {{- $value := .value -}} {{- if ne (printf "%T" $value) "string" -}} {{- errorf "shortcode %q: %s must be a string at %s" $name $parameter $position -}} {{- end -}} {{- $raw := printf "%s" $value -}} {{- if not $raw -}} {{- errorf "shortcode %q: %s must not be empty at %s" $name $parameter $position -}} {{- end -}} {{- if findRE `[\x00-\x20\x7f]` $raw -}} {{- errorf "shortcode %q: %s must not contain whitespace or control characters at %s" $name $parameter $position -}} {{- end -}} {{- if hasPrefix $raw "//" -}} {{- errorf "shortcode %q: %s must not use a protocol-relative URL at %s" $name $parameter $position -}} {{- end -}} {{- $parsedResult := try (urls.Parse $raw) -}} {{- with $parsedResult.Err -}} {{- errorf "shortcode %q: invalid %s %q at %s: %s" $name $parameter $raw $position . -}} {{- end -}} {{- $parsed := $parsedResult.Value -}} {{- $scheme := lower $parsed.Scheme -}} {{- $allowed := cond (eq $kind "image") (slice "http" "https") (slice "http" "https" "mailto") -}} {{- if and $scheme (not (in $allowed $scheme)) -}} {{- errorf "shortcode %q: unsupported %s scheme %q at %s" $name $parameter $scheme $position -}} {{- end -}} {{- if and (in (slice "http" "https") $scheme) (not $parsed.Host) -}} {{- errorf "shortcode %q: %s %q requires a host at %s" $name $parameter $raw $position -}} {{- end -}} {{- if and (eq $scheme "mailto") (not (or $parsed.Opaque $parsed.Path)) -}} {{- errorf "shortcode %q: %s %q requires an address at %s" $name $parameter $raw $position -}} {{- end -}} {{- if and (eq $kind "image") (not $scheme) (not $parsed.Path) -}} {{- errorf "shortcode %q: %s %q requires an image path at %s" $name $parameter $raw $position -}} {{- end -}} {{- $href := $raw -}} {{- if not $scheme -}} {{- if not (or (hasPrefix $raw "#") (hasPrefix $raw "?")) -}} {{- $href = strings.TrimPrefix "/" $raw | relURL -}} {{- end -}} {{- end -}} {{- $markdown := replace $href `\` `\\` -}} {{- $markdown = replace $markdown "(" `\(` -}} {{- $markdown = replace $markdown ")" `\)` -}} {{- return (dict "href" $href "markdown" $markdown "external" (ne $scheme "")) -}}
