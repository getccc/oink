false -}} {{- $alt := "" -}} {{- $decorative := false -}} {{- $matchMode := "legacy" -}} {{- if $named -}} {{- $matchMode = "exact" -}} {{- $allowed := slice "src" "command" "options" "alt" "decorative" -}} {{- range $param, $_ := .Params -}} {{- if not (in $allowed $param) -}} {{- errorf "shortcode %q: unsupported parameter %q at %s" $.Name $param $.Position -}} {{- end -}} {{- end -}} {{- range $required := slice "src" "command" "options" -}} {{- if not (isset $.Params $required) -}} {{- errorf "shortcode %q requires parameter %s at %s" $.Name $required $.Position -}} {{- end -}} {{- end -}} {{- $source = .Get "src" -}} {{- $command = .Get "command" -}} {{- $options = .Get "options" -}} {{- $hasAlt = isset .Params "alt" -}} {{- if $hasAlt }}{{ $alt = .Get "alt" }}{{ end -}} {{- if isset .Params "decorative" -}} {{- $decorativeValue := .Get "decorative" -}} {{- if ne (printf "%T" $decorativeValue) "bool" -}} {{- errorf "shortcode %q: decorative must be boolean at %s" .Name .Position -}} {{- end -}} {{- $decorative = $decorativeValue -}} {{- end -}} {{- else -}} {{- if ne (len .Params) 3 -}} {{- errorf "shortcode %q legacy form requires exactly src, command, and options at %s" .Name .Position -}} {{- end -}} {{- $source = .Get 0 -}} {{- $command = .Get 1 -}} {{- $options = .Get 2 -}} {{- end -}} {{- $caption := strings.TrimSpace (printf "%s" .InnerDeindent) -}} {{- $resolved := partial "content/image-resolve.html" (dict "name" .Name "position" .Position "page" .Page "src" $source "matchMode" $matchMode "hasAlt" $hasAlt "alt" $alt "decorative" $decorative "requireAlt" $named "caption" $caption ) -}} {{- $image := partial "content/image-process.html" (dict "name" .Name "position" .Position "image" $resolved "command" $command "options" $options ) -}} {{- $outputFormat := lower (.Page.Store.Get "tdOutputFormat" | default "html") -}} {{- if eq $outputFormat "markdown" -}} {{- $markdownAlt := partial "content/markdown-escape.html" $image.alt -}} {{- $markdown := printf "![%s](%s)" $markdownAlt $image.markdownSrc -}} {{- with $caption }}{{ $markdown = printf "%s\n\n%s" $markdown . }}{{ end -}} {{- with $image.byline -}} {{- $markdown = printf "%s\n\n_%s_" $markdown (partial "content/markdown-escape.html" .) -}} {{- end -}} {{- $markdown | safeHTML -}} {{- else -}} {{- if not $named -}} {{- else -}} {{- end -}} {{- end -}}
