{{- /* Collect one raw-code child for the enclosing code-group. */ -}} {{- with .Parent -}} {{- if ne $.Parent.Name "code-group" -}} {{- errorf "shortcode %q must be enclosed by code-group, not %q, at %s" $.Name $.Parent.Name $.Position -}} {{- end -}} {{- else -}} {{- errorf "shortcode %q must be enclosed by code-group at %s" $.Name $.Position -}} {{- end -}} {{- if not .IsNamedParams -}} {{- errorf "shortcode %q accepts named parameters only at %s" .Name .Position -}} {{- end -}} {{- $allowed := slice "title" "value" "lang" "selected" "copy" "wrap" "collapse" "lineNos" "lineNoStart" "hl_lines" "tabWidth" "anchorLineNos" "style" -}} {{- range $name, $_ := .Params -}} {{- if not (in $allowed $name) -}} {{- errorf "shortcode %q: unsupported parameter %q at %s" $.Name $name $.Position -}} {{- end -}} {{- end -}} {{- $titleParam := .Get "title" -}} {{- if not (isset .Params "title") -}} {{- errorf "shortcode %q requires parameter title at %s" .Name .Position -}} {{- end -}} {{- $title := strings.TrimSpace (printf "%v" $titleParam) -}} {{- if not $title -}} {{- errorf "shortcode %q: title must not be empty at %s" .Name .Position -}} {{- end -}} {{- $valueParam := .Get "value" -}} {{- if not (isset .Params "value") -}} {{- errorf "shortcode %q requires parameter value at %s" .Name .Position -}} {{- end -}} {{- $value := strings.TrimSpace (printf "%v" $valueParam) -}} {{- if not (findRE `^[a-z0-9][a-z0-9_-]*$` $value) -}} {{- errorf "shortcode %q: value must match ^[a-z0-9][a-z0-9_-]*$; got %q at %s" .Name $value .Position -}} {{- end -}} {{- $language := strings.TrimSpace (printf "%v" (.Get "lang" | default "text")) -}} {{- if not (findRE `^[A-Za-z0-9_+.-]+$` $language) -}} {{- errorf "shortcode %q: lang must be a single lexer token; got %q at %s" .Name $language .Position -}} {{- end -}} {{- $selected := false -}} {{- $selectedParam := .Get "selected" -}} {{- if isset .Params "selected" -}} {{- if ne (printf "%T" $selectedParam) "bool" -}} {{- errorf "shortcode %q: selected must be boolean at %s" .Name .Position -}} {{- end -}} {{- $selected = $selectedParam -}} {{- end -}} {{- $wrapParam := .Get "wrap" -}} {{- if and (isset .Params "wrap") (ne (printf "%T" $wrapParam) "bool") -}} {{- errorf "shortcode %q: wrap must be boolean at %s" .Name .Position -}} {{- end -}} {{- $attributeScratch := newScratch -}} {{- $attributeScratch.Set "values" (dict) -}} {{- range slice "copy" "wrap" "collapse" -}} {{- $parameter := $.Get . -}} {{- if isset $.Params . -}} {{- $attributeScratch.SetInMap "values" . $parameter -}} {{- end -}} {{- end -}} {{- $optionScratch := newScratch -}} {{- $optionScratch.Set "values" (dict) -}} {{- range slice "lineNos" "lineNoStart" "hl_lines" "tabWidth" "anchorLineNos" "style" -}} {{- $parameter := $.Get . -}} {{- if isset $.Params . -}} {{- $optionScratch.SetInMap "values" . $parameter -}} {{- end -}} {{- end -}} {{- $content := replaceRE `^\r?\n` "" .Inner -}} {{- $content = replaceRE `\r?\n[\t ]*$` "" $content -}} {{- $tab := dict "title" $title "value" $value "language" $language "selected" $selected "attributes" ($attributeScratch.Get "values") "options" ($optionScratch.Get "values") "content" $content "position" .Position -}} {{- .Parent.Scratch.SetInMap "codeTabs" (printf "%04d" .Ordinal) $tab -}}
